1. Introduction

1.1 Purpose

Nisol Intelligence is the AI-powered intelligence and advisory engine of the Nisol Platform. It analyzes assessment responses collected through Nisol Discovery and automatically generates executive insights, AI opportunities, implementation roadmaps, ROI estimates, solution blueprints, and consulting proposals.

Nisol Intelligence is not a standalone application. It is a module within the authenticated Nisol Platform and shares the same authentication, database, design system, and navigation as the public website and Nisol Discovery.

1.2 Product Vision

The Nisol Platform enables organizations to progress from AI assessment to AI transformation through a unified experience.

6.2 AI Client Abstraction

typescript

// lib/ai/client.ts

type ModelProvider = 'openai' | 'anthropic' | 'google';

interface ModelConfig {

provider: ModelProvider;

modelId: string;

apiKey: string;

}

class AIClient {

private models: ModelConfig[];

async generateWithFallback(

prompt: string,

preferredModels: string[]

): Promise<string> {

for (const model of preferredModels) {

try {

return await this.callModel(model, prompt);

} catch (error) {

if (this.isRetryableError(error)) {

continue; // Try next model

}

throw error;

}

}

throw new Error('All models failed');

}

}

-- ============================================

CREATE TABLE public.audit_reports (

id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

-- Report metadata

version INT DEFAULT 1,

-- 10 Core Outputs (stored as JSON/Text)

executive_summary TEXT,

ai_readiness_assessment JSONB, -- { overall_score: 3.8, radar_data: {...} }

capability_scores JSONB, -- { "Leadership": 3.5, "Data": 4.2, ... }

roi_estimates JSONB, -- { cost_savings: 500000, revenue_uplift: 250000, payback_months: 6 }

proposal_draft TEXT,

-- Metadata

generated_at TIMESTAMPTZ DEFAULT now(),

last_edited_at TIMESTAMPTZ DEFAULT now(),

finalized_at TIMESTAMPTZ,
